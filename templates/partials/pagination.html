{% macro pagination(page, total_pages, endpoint, query=None, per_page=None, target=None, total=None, filters=None) -%} {% if total is not none or total_pages > 1 %} {% set start = [1, page - 2]|max %} {% set end = [total_pages, page + 2]|min %} {% set extra = namespace(qs="") %} {% if query %}{% set extra.qs = extra.qs ~ "&q=" ~ query|urlencode %}{% endif %} {% if per_page %}{% set extra.qs = extra.qs ~ "&per_page=" ~ per_page %}{% endif %} {% if filters %} {% for key, value in filters.items() %} {% if value %}{% set extra.qs = extra.qs ~ "&" ~ key ~ "=" ~ value|string|urlencode %}{% endif %} {% endfor %} {% endif %} {% set qs = extra.qs %}
{% if total is not none %}{{ total }} {{ t("pagination.total") }} · {% endif %}{{ t("pagination.page_of").format(page=page, total=total_pages) }}
{% if page > 1 %} {% set url = endpoint ~ "?page=" ~ (page - 1) ~ qs %} {{ t("pagination.previous") }} {% endif %} {% for p in range(start, end + 1) %} {% set url = endpoint ~ "?page=" ~ p ~ qs %} {{ p }} {% endfor %} {% if page < total_pages %} {% set url = endpoint ~ "?page=" ~ (page + 1) ~ qs %} {{ t("pagination.next") }} {% endif %}
{% endif %} {%- endmacro %}
